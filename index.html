// Convolution Engine
void convolve(const Image *src, Image *dst,
              const Kernel *k) {
  int R = k->size / 2;
  for (int y = 0; y < src->h; ++y) {
    for (int x = 0; x < src->w; ++x) {
      int acc = 0, idx = 0;
      for (int dy = -R; dy <= R; ++dy)
        for (int dx = -R; dx <= R; ++dx)
          acc += src->data[clamp(y+dy) * src->w
               + clamp(x+dx)] * k->w[idx++];
      dst->data[y*dst->w + x]
        = clamp_u8(acc / k->norm + k->bias);
    }
  }
}
